schema {
  query: MappingQuery
}

type MappingQuery {
  child(id: ID, ids: [ID!], where: MappingChildWhere): MappingChild!
  childFirst(id: ID, ids: [ID!], where: MappingChildWhere): MappingChild!
  children(id: ID, ids: [ID!], where: MappingChildWhere, orderBy: [MappingChildOrderBy!], skip: Int, take: Int): [MappingChild!]!
  childrenOmitQueryArguments: [MappingChild!]!
  parent(id: ID, ids: [ID!], where: MappingParentWhere): MappingParent!
  parentFirst(id: ID, ids: [ID!], where: MappingParentWhere): MappingParent!
  parents(id: ID, ids: [ID!], where: MappingParentWhere, orderBy: [MappingParentOrderBy!], skip: Int, take: Int): [MappingParent!]!
}

type MappingChild {
  parent: MappingParent!
  id: ID!
  parentId: ID!
  property: String
}

type MappingParent {
  children(id: ID, ids: [ID!], where: MappingChildWhere, orderBy: [MappingChildOrderBy!], skip: Int, take: Int): [MappingChild!]!
  id: ID!
  jsonProperty: [String!]!
  property: String
}

input MappingChildWhere {
  and: [MappingChildWhere!]
  or: [MappingChildWhere!]
  not: MappingChildWhere
  id: GuidComparison
  parent: MappingParentWhere
  parentId: GuidComparison
  property: StringComparison
}

input GuidComparison {
  equal: ID
  notEqual: ID
  in: [ID]
  greaterThan: ID
  greaterThanOrEqual: ID
  lessThan: ID
  lessThanOrEqual: ID
}

input MappingParentWhere {
  and: [MappingParentWhere!]
  or: [MappingParentWhere!]
  not: MappingParentWhere
  children: MappingChildCollectionWhere
  id: GuidComparison
  ignoreByName: StringComparison
  property: StringComparison
}

input MappingChildCollectionWhere {
  any: MappingChildWhere
  all: MappingChildWhere
  none: MappingChildWhere
}

input StringComparison {
  equal: String
  notEqual: String
  in: [String]
  startsWith: String
  endsWith: String
  contains: String
  like: String
}

input MappingChildOrderBy {
  id: SortDirection
  parent: MappingParentOrderBy
  parentId: SortDirection
  property: SortDirection
}

enum SortDirection {
  ascending
  descending
}

input MappingParentOrderBy {
  id: SortDirection
  ignoreByName: SortDirection
  property: SortDirection
}
